function multisignMemo(memoClaims, signerKeyRefs[], sigAlgo):
    for keyRef in signerKeyRefs:
        // Optionally require each signer to sign the canonical memo independently;
        // or require sequential signing and include prior signatures in canonicalization,
        // but simplest: each signs the same canonical memo.
